# Nintendont ARM kernel. Links against the prebuilt big-endian be/lib{c,gcc}.a,
# fatfs-arm, and the generated kernel/asm + codehandler blob headers. Produces
# kernel.elf -> kernel.bin -> data/kernel.zip (the loader unzips it at boot).
# The susamune mod is injected via susamune_inject.h (found on the include path,
# pulled in by Patch.c under __has_include). Was kernel/Makefile.

set(_sources
    start.s common.c alloc.c GCAM.c JVSIO.c JVSIOMessage.c FST.c DI.c RealDI.c
    Patch.c PatchTimers.c TRI.c PatchWidescreen.c ISO.c Stream.c adp.c
    EXI.c SRAM.c GCNCard.c SI.c HID.c diskio.c Config.c utils_asm.S ES.c NAND.c
    main.c syscalls.s ReadSpeed.c vsprintf.c string.c prs.c
    SDI.c usb.c usbstorage.c wdvd.c
    lwbt/bte.c lwbt/btmemb.c lwbt/btmemr.c lwbt/btpbuf.c lwbt/hci.c lwbt/l2cap.c
    lwbt/physbusif.c BT.c)

add_executable(kernel ${_sources})
set_target_properties(kernel PROPERTIES OUTPUT_NAME kernel SUFFIX ".elf")

target_compile_options(kernel PRIVATE
    -mbig-endian -O2 -std=gnu89 -g -Wall
    -mcpu=arm926ej-s -msoft-float -mfloat-abi=soft
    -fdata-sections -ffunction-sections
    -fno-builtin-toupper -fno-builtin-memcpy -fno-builtin-memset
    -fno-builtin-memcmp -fno-builtin-malloc -fno-builtin-free
    $<$<COMPILE_LANGUAGE:ASM>:-D_LANGUAGE_ASSEMBLY>)

target_compile_definitions(kernel PRIVATE
    $<$<BOOL:${LAUNCHER_DISABLE_SUSAMUNE_PATCH}>:LAUNCHER_DISABLE_SUSAMUNE_PATCH>)

target_include_directories(kernel PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}"
    "${CMAKE_CURRENT_SOURCE_DIR}/../../include"
    "${CMAKE_CURRENT_SOURCE_DIR}/../fatfs"
    "${LAUNCHER_GEN_DIR}"          # codehandler.h + asm/*.h blob headers
    "${SUSAMUNE_INJECT_DIR}")      # susamune_inject.h (mod blob + writes)

# Recompile Patch.o when the mod's inject header changes -- this is the
# fine-grained dependency that lets a mod-source edit relink only the kernel.
set_source_files_properties(Patch.c PROPERTIES
    OBJECT_DEPENDS "${SUSAMUNE_INJECT_DIR}/susamune_inject.h")

target_link_libraries(kernel PRIVATE
    fatfs-arm
    "${CMAKE_CURRENT_SOURCE_DIR}/be/libc.a"
    "${CMAKE_CURRENT_SOURCE_DIR}/be/libgcc.a")

target_link_options(kernel PRIVATE
    -nostartfiles -nodefaultlibs -mbig-endian -n
    -Wl,--gc-sections
    "-Wl,-T,${CMAKE_CURRENT_SOURCE_DIR}/kernel.ld"
    "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/kernel.map")
set_target_properties(kernel PROPERTIES
    LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/kernel.ld")

# kernel headers are generated by the ppc group (separate build); this ordering
# is guaranteed by the super-build's ExternalProject DEPENDS.

set(_kernel_bin "${CMAKE_CURRENT_BINARY_DIR}/kernel.bin")
add_custom_command(
    OUTPUT "${LAUNCHER_DATA_DIR}/kernel.zip"
    COMMAND "${CMAKE_COMMAND}" -E make_directory "${LAUNCHER_DATA_DIR}"
    COMMAND "${XPREFIX}objcopy${XSUFFIX}" -S -O binary "$<TARGET_FILE:kernel>" "${_kernel_bin}"
    COMMAND "${CMAKE_COMMAND}" -E chdir "${CMAKE_CURRENT_BINARY_DIR}"
            "${CMAKE_COMMAND}" -E tar cf "${LAUNCHER_DATA_DIR}/kernel.zip" --format=zip -- kernel.bin
    DEPENDS kernel
    COMMENT "packaging kernel.zip"
    VERBATIM)
add_custom_target(kernel_zip ALL DEPENDS "${LAUNCHER_DATA_DIR}/kernel.zip")
